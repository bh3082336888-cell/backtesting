**Goal:** `lab judge` decides promote or kill for a strategy plugin against its parent with no human involved, gives each family at most one exam on the locked holdout under a global cap, and gates paper approval on a passed exam.

**Architecture:** A new `perp_lab/judge/` package with one job per file: `gates.py` (pure math), `budget.py` (the only code that unlocks the holdout), `exam.py` (the in-process holdout run) and `judge.py` (orchestration and public verdicts). **Tech Stack:** Python 3.12, pandas 3, numpy, SQLAlchemy Core (SQLite), Typer, Streamlit, pytest. No new dependencies.

- No new dependencies: normal CDF via `statistics.NormalDist`, file locking via `fcntl`.
- Leave `aiohttp<3.14` in `pyproject.toml` and `holdout_start: "2026-07-01"` in `config/lab.yaml` untouched.

```python
"""Causality (lookahead) check, usable at judge time as well as in the test suite.

For several cut points k: features computed on bars[:k] must equal features computed on
the full series for rows < k, and the engine must emit identical signals for bars < k-1.
Any use of future data breaks one of the two.
"""

from __future__ import annotations

import numpy as np
import pandas as pd

from perp_lab.engine.backtest import EngineParams, run_backtest

H = 3_600_000
CUTS = [400, 1100, 1777, 2500]


def synthetic_bars(n=3000, seed=0, start="2023-01-01", funding_every=8, tf_ms=H):
    """Synthetic perp bars with regimes, trends and funding extremes."""
    rng = np.random.default_rng(seed)
    drift = np.repeat(rng.normal(0, 0.002, n // 200 + 1), 200)[:n]
    r = drift + rng.normal(0, 0.01, n)
    close = 100 * np.exp(np.cumsum(r))
    open_ = np.r_[100.0, close[:-1]]
    hi = np.maximum(open_, close) * (1 + np.abs(rng.normal(0, 0.004, n)))
    lo = np.minimum(open_, close) * (1 - np.abs(rng.normal(0, 0.004, n)))
    t0 = int(pd.Timestamp(start, tz="UTC").value // 10**6)
    ot = t0 + np.arange(n) * tf_ms
    fc = np.zeros(n)
    idx = np.arange(0, n, funding_every)
    fc[idx] = 0.0001 + rng.normal(0, 0.00005, len(idx))
    spikes = rng.choice(idx, size=max(len(idx) // 15, 1), replace=False)
    fc[spikes] = rng.choice([-1, 1], size=len(spikes)) * rng.uniform(0.0008, 0.002, len(spikes))
    df = pd.DataFrame({
        "open_time": ot, "open": open_, "high": hi, "low": lo, "close": close,
        "volume": rng.uniform(100, 1000, n), "quote_volume": np.nan, "trades": np.nan,
        "taker_buy_volume": np.nan, "taker_buy_quote_volume": np.nan, "close_time": ot + tf_ms - 1,
        "funding_charge": fc, "funding_last": pd.Series(np.where(fc != 0, fc, np.nan)).ffill().values,
        "funding_interval_h": 8.0, "oi": np.nan, "oi_value": np.nan,
        "regime": rng.choice(["trending", "ranging", "high_vol"], n), "regime_adx": 20.0,
    })
    df.index = pd.to_datetime(df.open_time, unit="ms", utc=True)
    return df


def assert_causal(strat_factory, bars):
    full = strat_factory()
    f_full = full.prepare(bars)
    assert f_full.index.equals(bars.index), "prepare() must return features on the bars index"
    r_full = run_backtest(full, bars, f_full, symbol="X", tf="1h", ep=EngineParams(), record_signals=True)
    for k in CUTS:
        cut = bars.iloc[:k]
        s = strat_factory()
        f_cut = s.prepare(cut)
        pd.testing.assert_frame_equal(f_full.iloc[:k], f_cut, check_exact=False, rtol=1e-9, atol=1e-12,
                                      obj=f"features at cut {k}")
        r_cut = run_backtest(s, cut, f_cut, symbol="X", tf="1h", ep=EngineParams(), record_signals=True)
        a = [x for x in r_full.signals if x[0] < k - 1]
        b = [x for x in r_cut.signals if x[0] < k - 1]
        assert a == b, f"signals diverge before cut {k}"


def causality_failures(cls, bars: pd.DataFrame | None = None) -> list[str]:
    """assert_causal on the default params and every grid corner; one line per failing variant.

    A crash on synthetic data disqualifies a plugin as surely as a peek at the future."""
    bars = synthetic_bars(seed=1) if bars is None else bars
    variants = [{}] + [{k: v} for k, vals in cls.grid.items() for v in (vals[0], vals[-1])]
    out = []
    for kw in variants:
        try:
            assert_causal(lambda: cls(**kw), bars)
        except Exception as e:
            label = ", ".join(f"{k}={v}" for k, v in kw.items()) or "defaults"
            msg = str(e).splitlines()[0] if str(e) else ""
            out.append(f"{label}: {type(e).__name__}: {msg}" if msg else f"{label}: {type(e).__name__}")
    return out
```

```python
"""Judge gates: pure functions, no I/O. Resampling is seeded, so a judgement is reproducible."""

from __future__ import annotations

import hashlib
import math
from statistics import NormalDist

import numpy as np
import pandas as pd

from perp_lab.engine.metrics import sharpe

EULER_GAMMA = 0.5772156649
_N = NormalDist()


def seed_from(run_id: str) -> int:
    return int(hashlib.sha256(run_id.encode()).hexdigest()[:8], 16)


def moments(r: pd.Series) -> tuple[float, int, float, float]:
    """Daily Sharpe (ddof=1), T, skew and Pearson kurtosis (population moments)."""
    x = r.dropna().to_numpy(dtype=float)
    t = len(x)
    if t < 3 or x.std(ddof=1) == 0:
        return 0.0, t, 0.0, 3.0
    d = x - x.mean()
    m2, m3, m4 = (d ** 2).mean(), (d ** 3).mean(), (d ** 4).mean()
    return float(x.mean() / x.std(ddof=1)), t, float(m3 / m2 ** 1.5), float(m4 / m2 ** 2)


def expected_max_sharpe(n_trials: int, var_trials: float) -> float:
    """SR0: the daily Sharpe the best of n_trials unskilled attempts reaches by luck."""
    if n_trials < 2 or var_trials <= 0:
        return 0.0
    return math.sqrt(var_trials) * ((1 - EULER_GAMMA) * _N.inv_cdf(1 - 1 / n_trials)
                                    + EULER_GAMMA * _N.inv_cdf(1 - 1 / (n_trials * math.e)))


def dsr_from_moments(sr: float, t: int, skew: float, kurt: float, n_trials: int, var_trials: float) -> float:
    """Deflated Sharpe Ratio (Bailey & Lopez de Prado, 2014)."""
    if t < 3:
        return 0.0
    den = 1 - skew * sr + (kurt - 1) / 4 * sr ** 2
    if den <= 0:
        return 0.0
    sr0 = expected_max_sharpe(n_trials, var_trials)
    return float(_N.cdf((sr - sr0) * math.sqrt(t - 1) / math.sqrt(den)))


def deflated_sharpe(r: pd.Series, n_trials: int, var_trials: float) -> dict:
    sr, t, skew, kurt = moments(r)
    flat = t < 3 or float(r.dropna().std(ddof=1)) == 0
    return {"dsr": 0.0 if flat else dsr_from_moments(sr, t, skew, kurt, n_trials, var_trials),
            "sr0": expected_max_sharpe(n_trials, var_trials), "sr": sr, "t": t, "skew": skew, "kurt": kurt}


def block_indices(n: int, length: int, block: int, n_boot: int, rng: np.random.Generator) -> np.ndarray:
    """(n_boot, length) moving-block bootstrap indices into a series of n points."""
    block = max(1, min(block, n))
    k = -(-length // block)
    starts = rng.integers(0, n - block + 1, size=(n_boot, k))
    return (starts[:, :, None] + np.arange(block)[None, None, :]).reshape(n_boot, k * block)[:, :length]


def _row_sharpes(a: np.ndarray) -> np.ndarray:
    sd = a.std(axis=1, ddof=1)
    ok = sd > 0
    return np.where(ok, a.mean(axis=1) / np.where(ok, sd, 1.0), 0.0)


def beats_parent(child: pd.Series, parent: pd.Series, *, block: int, n_boot: int, seed: int) -> float:
    """Fraction of paired block-bootstrap resamples in which the child's Sharpe is strictly higher.

    Days are resampled jointly so the two series stay matched. Ties are not wins, so identical
    series score 0. Fewer than two blocks of overlapping days scores 0."""
    both = pd.concat({"c": child, "p": parent}, axis=1, join="inner").dropna()
    n = len(both)
    if n < 2 * block:
        return 0.0
    idx = block_indices(n, n, block, n_boot, np.random.default_rng(seed))
    c, p = both.c.to_numpy()[idx], both.p.to_numpy()[idx]
    return float((_row_sharpes(c) > _row_sharpes(p)).mean())


def holdout_consistent(holdout: pd.Series, n_trades: int, wf_oos: pd.Series, *, min_trades: int,
                       floor_pct: float, block: int, n_boot: int, seed: int) -> tuple[bool, list[str], dict]:
    """Pass iff enough trades, Sharpe > 0, and the holdout's cumulative return is at least the
    floor_pct percentile of same-length cumulative returns resampled from the walk-forward OOS record."""
    h = holdout.dropna()
    w = wf_oos.dropna().to_numpy(dtype=float)
    detail = {"days": len(h), "trades": int(n_trades), "sharpe": sharpe(h),
              "cum_return": float((1 + h).prod() - 1), "floor": None}
    reasons = []
    if n_trades < min_trades:
        reasons.append(f"not enough evidence: {n_trades} holdout trades < {min_trades}")
    if detail["sharpe"] <= 0:
        reasons.append(f"holdout Sharpe {detail['sharpe']:.2f} <= 0")
    if len(h) == 0 or len(w) < block:
        reasons.append("not enough data to compare with the walk-forward record")
    else:
        idx = block_indices(len(w), len(h), block, n_boot, np.random.default_rng(seed))
        detail["floor"] = float(np.percentile(np.prod(1 + w[idx], axis=1) - 1, floor_pct))
        if detail["cum_return"] < detail["floor"]:
            reasons.append(f"holdout return {detail['cum_return']:.1%} below the walk-forward "
                           f"p{floor_pct:g} for {len(h)} days ({detail['floor']:.1%})")
    return not reasons, reasons, detail
```

```python
# ---------------------------------------------------------------- judgements (Phase 3)

def _row(r) -> dict | None:
    return dict(r._mapping) if r is not None else None


def record_judgement(cfg, row: dict) -> int:
    """Insert one judgement. A second judgement of the same (strategy_hash, timeframe), e.g. two
    judges racing, returns the first one's id instead of failing."""
    if row["decision"] not in DECISIONS:
        raise ValueError(f"decision must be one of {sorted(DECISIONS)}")
    if row["holdout_status"] not in HOLDOUT_STATUSES:
        raise ValueError(f"holdout_status must be one of {sorted(HOLDOUT_STATUSES)}")
    now = _now()
    try:
        with engine(cfg).begin() as c:
            return int(c.execute(judgements.insert().values(**row, created_at=now, updated_at=now))
                       .inserted_primary_key[0])
    except sa.exc.IntegrityError:
        existing = get_judgement(cfg, row["strategy_hash"], row["timeframe"])
        if existing is None:
            raise
        return existing["id"]


def _judgement_where(cfg, where) -> dict | None:
    with engine(cfg).connect() as c:
        return _row(c.execute(sa.select(judgements).where(where)).first())


def get_judgement(cfg, strategy_hash: str, timeframe: str) -> dict | None:
    return _judgement_where(cfg, sa.and_(judgements.c.strategy_hash == strategy_hash,
                                         judgements.c.timeframe == timeframe))


def get_judgement_by_id(cfg, jid: int) -> dict | None:
    return _judgement_where(cfg, judgements.c.id == jid)


def judgement_for_run(cfg, run_id: str) -> dict | None:
    return _judgement_where(cfg, judgements.c.child_run_id == run_id)


def update_holdout(cfg, jid: int, status: str, reason: str, holdout_run_id: str | None = None) -> None:
    if status not in HOLDOUT_STATUSES:
        raise ValueError(f"holdout_status must be one of {sorted(HOLDOUT_STATUSES)}")
    vals = {"holdout_status": status, "holdout_reason": reason, "updated_at": _now()}
    if holdout_run_id is not None:
        vals["holdout_run_id"] = holdout_run_id
    with engine(cfg).begin() as c:
        c.execute(judgements.update().where(judgements.c.id == jid).values(**vals))


def pending_judgements(cfg) -> list[dict]:
    with engine(cfg).connect() as c:
        rows = c.execute(sa.select(judgements).where(judgements.c.holdout_status == "pending")
                         .order_by(judgements.c.created_at, judgements.c.id))
        return [dict(r._mapping) for r in rows]


def run_row(cfg, run_id: str) -> dict | None:
    with engine(cfg).connect() as c:
        return _row(c.execute(sa.select(runs).where(runs.c.run_id == run_id)).first())


def latest_wf_run(cfg, strategy: str, timeframe: str, strategy_hash: str, universe: list[str]) -> dict | None:
    """Most recent walk-forward run of exactly this plugin version, timeframe and symbol universe."""
    with engine(cfg).connect() as c:
        rows = c.execute(sa.select(runs).where(sa.and_(
            runs.c.strategy == strategy, runs.c.timeframe == timeframe,
            runs.c.strategy_hash == strategy_hash, runs.c.validation_mode == "walkforward"))
            .order_by(runs.c.created_at.desc())).all()
    for r in rows:
        if list(r.universe or []) == list(universe):
            return dict(r._mapping)
    return None


def family_of(cfg, strategy: str) -> str | None:
    """A plugin name's family: that of its first judgement, else of its first walk-forward run.
    Looked up by name so editing or re-running a root never starts a new family."""
    with engine(cfg).connect() as c:
        j = c.execute(sa.select(judgements.c.lineage_id).where(sa.and_(
            judgements.c.strategy == strategy, judgements.c.lineage_id.isnot(None)))
            .order_by(judgements.c.created_at, judgements.c.id).limit(1)).first()
        if j:
            return j[0]
        r = c.execute(sa.select(runs.c.lineage_id).where(sa.and_(
            runs.c.strategy == strategy, runs.c.validation_mode == "walkforward",
            runs.c.lineage_id.isnot(None))).order_by(runs.c.created_at).limit(1)).first()
    return r[0] if r else None


def trial_stats(cfg) -> tuple[int, float]:
    """N = distinct (strategy_hash, timeframe) among walk-forward runs, store-wide; V = sample
    variance of the daily Sharpes of the latest run of each (0.0 when N < 2)."""
    with engine(cfg).connect() as c:
        df = pd.read_sql(sa.select(runs.c.strategy_hash, runs.c.timeframe, runs.c.sharpe, runs.c.created_at)
                         .where(sa.and_(runs.c.validation_mode == "walkforward", runs.c.sharpe.isnot(None))), c)
    df = df.dropna(subset=["strategy_hash", "timeframe"])
    if df.empty:
        return 0, 0.0
    latest = df.sort_values("created_at").groupby(["strategy_hash", "timeframe"]).tail(1)
    n = len(latest)
    v = float((latest.sharpe / np.sqrt(365)).var(ddof=1)) if n >= 2 else 0.0
    return n, v
```

```python
"""Holdout budget: the only code allowed to unlock the holdout.

The audit log written by issue_holdout_token is the record of reads. One read = one distinct
judgement_id (entries without one count by nonce). Under an exclusive file lock: a judgement
that already holds a read may retry (its result was never published); otherwise the exam
window must be at least min_holdout_days, total reads stay under holdout_max_reads, and each
family gets one read. A missing or malformed log refuses; it never means "zero reads".
"""

from __future__ import annotations

import fcntl
import json
from contextlib import contextmanager
from dataclasses import dataclass
from pathlib import Path

from perp_lab.holdout import HoldoutToken, issue_holdout_token


class BudgetLogError(RuntimeError):
    pass


@dataclass(frozen=True)
class BudgetCheck:
    status: str     # mint | pending | blocked
    code: str       # ok | retry | too_short | budget_spent | family_examined


def log_path(cfg) -> Path:
    return Path(cfg["judge"]["holdout_log"])


def init_log(cfg) -> Path:
    p = log_path(cfg)
    if p.exists():
        raise FileExistsError(f"{p} already exists; the holdout read record is never reset")
    p.parent.mkdir(parents=True, exist_ok=True)
    p.touch()
    return p


def read_log(path: Path) -> list[dict]:
    if not path.exists():
        raise BudgetLogError(f"holdout log {path} is missing; create it once with `lab holdout-init`")
    entries = []
    for i, line in enumerate(path.read_text().splitlines(), start=1):
        if not line.strip():
            continue
        try:
            e = json.loads(line)
        except json.JSONDecodeError as ex:
            raise BudgetLogError(f"holdout log {path} line {i} is not JSON: {ex}") from None
        if not isinstance(e, dict) or "lineage_id" not in e or "nonce" not in e:
            raise BudgetLogError(f"holdout log {path} line {i} is not a holdout read entry")
        entries.append(e)
    return entries


def _read_key(e: dict) -> tuple:
    return ("judgement", e["judgement_id"]) if e.get("judgement_id") is not None else ("nonce", e["nonce"])


def reads_used(cfg) -> int:
    return len({_read_key(e) for e in read_log(log_path(cfg))})


def check(cfg, entries: list[dict], lineage_id: str, judgement_id: int, window_days: int) -> BudgetCheck:
    jc = cfg["judge"]
    if any(e.get("judgement_id") == judgement_id for e in entries):
        return BudgetCheck("mint", "retry")
    if window_days < jc["min_holdout_days"]:
        return BudgetCheck("pending", "too_short")
    if len({_read_key(e) for e in entries}) >= jc["holdout_max_reads"]:
        return BudgetCheck("blocked", "budget_spent")
    if any(e["lineage_id"] == lineage_id for e in entries):
        return BudgetCheck("blocked", "family_examined")
    return BudgetCheck("mint", "ok")


@contextmanager
def _locked(path: Path):
    path.parent.mkdir(parents=True, exist_ok=True)
    with open(f"{path}.lock", "a") as f:
        fcntl.flock(f, fcntl.LOCK_EX)
        try:
            yield
        finally:
            fcntl.flock(f, fcntl.LOCK_UN)


def mint(cfg, lineage_id: str, judgement_id: int, window_days: int) -> tuple[BudgetCheck, HoldoutToken | None]:
    """Check the budget and, if allowed, unlock the holdout. The read is logged before any data is read."""
    p = log_path(cfg)
    with _locked(p):
        chk = check(cfg, read_log(p), lineage_id, judgement_id, window_days)
        if chk.status != "mint":
            return chk, None
        return chk, issue_holdout_token(lineage_id, f"judge exam, judgement {judgement_id}", p,
                                        judgement_id=judgement_id)
```

```python
"""Holdout exam: one run with frozen parameters on the locked window, in the judge's process.

The token is checked by object identity and is invalid in any other process, so the exam never
uses the walk-forward worker pool. Indicators warm up on research data (each run starts at
history_start); only returns and trades inside [holdout_start, end) are kept.
"""

from __future__ import annotations

from dataclasses import dataclass
from pathlib import Path

import pandas as pd

from perp_lab.config import TF_MS
from perp_lab.data.store import load_bars
from perp_lab.engine.backtest import EngineParams, run_backtest
from perp_lab.engine.metrics import daily_returns, portfolio_returns
from perp_lab.holdout import check_token
from perp_lab.strategy import load_strategy_class

DEFAULT_FUNDING_H = 8.0


@dataclass
class ExamResult:
    start: pd.Timestamp
    end: pd.Timestamp
    params: dict
    daily: dict[str, pd.Series]
    portfolio: pd.Series
    trades: pd.DataFrame


def _files(d: Path) -> list[Path]:
    files = sorted(d.glob("*.parquet"))
    if not files:
        raise FileNotFoundError(f"no parquet files in {d}; run `lab backfill`")
    return files


def window_end(cfg, symbols: list[str], tf: str) -> pd.Timestamp:
    """The last 00:00 UTC up to which every symbol has closed candles and funding.

    Reads timestamps only (open_time, funding_time, interval_hours), never prices."""
    ends = []
    for s in symbols:
        last_open = max(int(pd.read_parquet(f, columns=["open_time"]).open_time.max())
                        for f in _files(cfg.data_dir / s / tf))
        fund = (pd.concat([pd.read_parquet(f, columns=["funding_time", "interval_hours"])
                           for f in _files(cfg.data_dir / s / "funding")])
                .sort_values("funding_time").iloc[-1])
        ih = fund.interval_hours if pd.notna(fund.interval_hours) else DEFAULT_FUNDING_H
        ends.append(min(last_open + TF_MS[tf], int(fund.funding_time) + int(ih * 3_600_000)))
    return pd.Timestamp(min(ends), unit="ms", tz="UTC").floor("D")


def run_exam(cfg, name: str, tf: str, symbols: list[str], params: dict, token, end: pd.Timestamp) -> ExamResult:
    check_token(token)
    cls = load_strategy_class(name)
    start = cfg.holdout_start
    daily, parts = {}, []
    for s in symbols:
        bars = load_bars(cfg, s, tf, end=end, holdout=token)
        strat = cls(**params)
        res = run_backtest(strat, bars, strat.prepare(bars), symbol=s, tf=tf, ep=EngineParams.from_config(cfg, s))
        dr = daily_returns(res.equity, res.time)
        daily[s] = dr[(dr.index >= start) & (dr.index < end)] if len(dr) else dr
        t = pd.DataFrame(res.trades)
        if len(t):
            st = pd.to_datetime(t.signal_time, unit="ms", utc=True)
            parts.append(t[(st >= start) & (st < end)])
    trades = pd.concat(parts, ignore_index=True) if parts else pd.DataFrame()
    return ExamResult(start, end, dict(params), daily, portfolio_returns(daily), trades)
```

```python
"""The judge: promote or kill a strategy plugin against its parent, with no human involved.

Spec: docs/superpowers/specs/2026-10-16-judge-design.md. Flow: pre-checks -> idempotency ->
lookahead -> parent run -> child walk-forward -> gates -> decision -> judgement + lineage edge ->
(optional holdout exam via `attempt`) -> public verdict.
"""

from __future__ import annotations

import json
import logging
from pathlib import Path
from typing import Callable

import pandas as pd

from perp_lab.judge import gates
from perp_lab.store import db
from perp_lab.strategy import list_plugins, load_strategy_class, plugin_hash, plugin_parent
from perp_lab.validate.causality import causality_failures
from perp_lab.validate.pipeline import validate_strategy

log = logging.getLogger(__name__)

PARENT_BOOT = 5_000
VERDICT_KEYS = {"strategy", "timeframe", "strategy_hash", "lineage_id", "parent_run_id", "decision",
                "reasons", "dsr", "n_trials", "sr0", "parent_passed", "beats_parent_frac",
                "holdout_status", "holdout_reason", "walkforward"}


class JudgeRefused(Exception):
    """A pre-check failed; nothing was recorded."""


def oos_portfolio(run_dir) -> pd.Series:
    return pd.read_parquet(Path(run_dir) / "oos_daily.parquet")["PORTFOLIO"].dropna()


def run_summary(run_dir) -> dict:
    return json.loads((Path(run_dir) / "summary.json").read_text())


def verdict_path(cfg, strategy: str, tf: str, strategy_hash: str) -> Path:
    return Path(cfg["judge"]["verdicts_dir"]) / f"{strategy}@{tf}@{strategy_hash[:10]}.json"


def write_verdict(cfg, j: dict) -> Path:
    """The proposer's only window on a judgement. Holdout outcome as status + fixed code only:
    never a number computed from holdout data, never the holdout run's id."""
    wf = None
    if j["child_run_id"]:
        run_dir = db.run_row(cfg, j["child_run_id"])["run_dir"]
        wf = {"run_id": j["child_run_id"], "metrics": run_summary(run_dir).get("metrics"),
              "report": str(Path(run_dir) / "report.md")}
    out = {k: j[k] for k in VERDICT_KEYS - {"walkforward"}}
    out["walkforward"] = wf
    p = verdict_path(cfg, j["strategy"], j["timeframe"], j["strategy_hash"])
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text(json.dumps(out, indent=1, default=str))
    return p


def finish(cfg, jid: int) -> dict:
    j = db.get_judgement_by_id(cfg, jid)
    return {**j, "verdict_path": str(write_verdict(cfg, j))}


def judge(cfg, name: str, tf: str, symbols: list[str] | None = None, workers: int | None = None, *,
          validate: Callable = validate_strategy, attempt: Callable | None = None) -> dict:
    jc = cfg["judge"]
    parent = plugin_parent(name)
    if parent is not None:
        if parent == name:
            raise JudgeRefused(f"{name} names itself as its PARENT")
        if parent not in list_plugins():
            raise JudgeRefused(f"PARENT '{parent}' is not a plugin in strategies/")
    h = plugin_hash(name)
    if parent is not None and plugin_hash(parent) == h:
        raise JudgeRefused(f"{name} is byte-identical to its parent {parent}; not a variation")

    existing = db.get_judgement(cfg, h, tf)
    if existing:
        return {**existing, "verdict_path": str(verdict_path(cfg, name, tf, h)), "existing": True}

    family = db.family_of(cfg, parent if parent else name)
    fails = causality_failures(load_strategy_class(name))
    if fails:
        jid = db.record_judgement(cfg, {
            "strategy": name, "timeframe": tf, "strategy_hash": h, "child_run_id": None,
            "parent_run_id": None, "lineage_id": family, "decision": "kill",
            "reasons": [f"lookahead: {f}" for f in fails], "holdout_status": "n/a", "holdout_reason": "n/a"})
        return finish(cfg, jid)

    universe = symbols or cfg.symbols
    parent_row = None
    if parent is not None:
        parent_row = db.latest_wf_run(cfg, parent, tf, plugin_hash(parent), universe)
        if parent_row is None:
            prec, _ = validate(cfg, parent, tf, symbols, workers=workers, lineage_id=db.family_of(cfg, parent))
            parent_row = db.run_row(cfg, prec["run_id"])
        family = parent_row["lineage_id"]

    rec, _ = validate(cfg, name, tf, symbols, workers=workers,
                      parent_run_id=parent_row["run_id"] if parent_row else None, lineage_id=family)
    child = db.run_row(cfg, rec["run_id"])
    n, v = db.trial_stats(cfg)
    cr = oos_portfolio(child["run_dir"])
    d = gates.deflated_sharpe(cr, n, v)
    reasons = list(rec.get("fail_reasons") or [])
    if d["dsr"] < jc["dsr_min"]:
        reasons.append(f"Deflated Sharpe {d['dsr']:.2f} < {jc['dsr_min']} (N={n})")

    parent_passed, frac = None, None
    if parent_row is not None:
        pr = oos_portfolio(parent_row["run_dir"])
        parent_passed = bool(parent_row["verdict"] == "candidate"
                             and gates.deflated_sharpe(pr, n, v)["dsr"] >= jc["dsr_min"])
        if parent_passed:
            frac = gates.beats_parent(cr, pr, block=jc["bootstrap_block_days"], n_boot=PARENT_BOOT,
                                      seed=gates.seed_from(child["run_id"]))
            if frac < jc["parent_beat_frac"]:
                reasons.append(f"beats its passing parent in {frac:.0%} of resamples "
                               f"< {jc['parent_beat_frac']:.0%}")

    decision = "kill" if reasons else "promote"
    jid = db.record_judgement(cfg, {
        "strategy": name, "timeframe": tf, "strategy_hash": h, "child_run_id": child["run_id"],
        "parent_run_id": parent_row["run_id"] if parent_row else None, "lineage_id": child["lineage_id"],
        "decision": decision, "reasons": reasons, "dsr": d["dsr"], "n_trials": n, "sr0": d["sr0"],
        "parent_passed": parent_passed, "beats_parent_frac": frac,
        "holdout_status": "pending" if decision == "promote" else "n/a",
        "holdout_reason": "awaiting_exam" if decision == "promote" else "n/a"})
    if parent_row is not None:
        db.add_edge(cfg, parent_row["run_id"], child["run_id"], decision,
                    "; ".join(reasons) or "passed every gate", decided_by="harness")
    if decision == "promote" and attempt is not None:
        attempt(cfg, jid)
    return finish(cfg, jid)
```

```python
def _record_holdout_run(cfg, j: dict, s: dict, res, ok: bool, reasons: list[str], detail: dict) -> str:
    rid = new_run_id()
    rec = {"run_id": rid, "strategy": j["strategy"], "strategy_hash": j["strategy_hash"],
           "hypothesis": s.get("hypothesis"), "repo_commit": repo_commit(), "params": res.params,
           "universe": list(res.daily), "timeframes": [j["timeframe"]], "validation_mode": "holdout",
           "date_start": str(res.start.date()), "date_end": str(res.end.date()),
           "status": "passed" if ok else "failed", "fail_reasons": reasons,
           "metrics": {**summarize(res.portfolio, res.trades), "floor": detail["floor"]},
           "lineage_id": j["lineage_id"], "parent_run_id": None,
           "created_at": datetime.now(timezone.utc).isoformat(),
           "notes": f"holdout exam of {j['child_run_id']} (judgement {j['id']})"}
    daily = pd.DataFrame({**res.daily, "PORTFOLIO": res.portfolio})
    write_run(cfg, rec, {"holdout_trades": res.trades, "holdout_daily": daily})

def attempt_exam(cfg, jid: int, *, run: Callable = run_exam, end_fn: Callable = window_end) -> dict:
    """Try the holdout exam for a pending judgement, under the budget rules."""
    j = db.get_judgement_by_id(cfg, jid)
    if j is None or j["holdout_status"] != "pending":
        return j
    child = db.run_row(cfg, j["child_run_id"])
    s = run_summary(child["run_dir"])
    syms, tf = s["universe"], j["timeframe"]
    end = end_fn(cfg, syms, tf)
    days = max(0, (end - cfg.holdout_start).days)
    chk, token = budget.mint(cfg, j["lineage_id"], jid, days)
    if token is None:
        db.update_holdout(cfg, jid, chk.status, chk.code)
        return finish(cfg, jid)
    try:
        res = run(cfg, j["strategy"], tf, syms, s["sensitivity"]["base_params"], token, end)
    except Exception:
        db.update_holdout(cfg, jid, "pending", "exam_error")   # same judgement may retry on the same read
        raise
    jc = cfg["judge"]
    ok, reasons, detail = gates.holdout_consistent(
        res.portfolio, len(res.trades), oos_portfolio(child["run_dir"]),
        min_trades=jc["holdout_min_trades"], floor_pct=jc["holdout_floor_pct"],
        block=jc["bootstrap_block_days"], n_boot=HOLDOUT_BOOT, seed=gates.seed_from(child["run_id"]))
    hrid = _record_holdout_run(cfg, j, s, res, ok, reasons, detail)
    db.update_holdout(cfg, jid, "passed" if ok else "failed", "passed" if ok else "failed", hrid)
    return finish(cfg, jid)


def exam_pending(cfg, **kw) -> list[dict]:
    """Retry every pending exam, oldest first."""
    return [attempt_exam(cfg, j["id"], **kw) for j in db.pending_judgements(cfg)]
```
